  <script>

    const SCALE = 100; // 1 m = 100 px

    const GRID_CM = 25;

    const GRID_PX = GRID_CM; // weil 100 cm = 100 px


    let packageTemplates = [];

    let placedBoxes = [];

    let selectedId = null;

    let dragState = null;


    const colors = [

      'rgba(56,189,248,.78)',

      'rgba(34,197,94,.78)',

      'rgba(251,191,36,.78)',

      'rgba(168,85,247,.78)',

      'rgba(244,114,182,.78)',

      'rgba(45,212,191,.78)',

      'rgba(251,146,60,.78)'

    ];


    function trailerMeters() {

      return {

        length: Number(document.getElementById('trailerLength').value) || 4,

        width: Number(document.getElementById('trailerWidth').value) || 2.5,

      };

    }


    function trailerPx() {

      const t = trailerMeters();

      return { width: t.length * SCALE, height: t.width * SCALE };

    }


    function updateTrailer() {

      const trailer = document.getElementById('trailer');

      const tpx = trailerPx();

      trailer.style.width = `${tpx.width}px`;

      trailer.style.height = `${tpx.height}px`;

      validateAllBoxes();

      renderPlacedBoxes();

      updateStats();

    }


    function addPackageTemplate() {

      const name = document.getElementById('boxName').value.trim() || 'Paket';

      const count = Math.max(1, Number(document.getElementById('boxCount').value) || 1);

      const lengthCm = Math.max(1, Number(document.getElementById('boxLength').value) || 1);

      const widthCm = Math.max(1, Number(document.getElementById('boxWidth').value) || 1);

      const id = crypto.randomUUID();


      packageTemplates.push({

        id,

        name,

        count,

        lengthCm,

        widthCm,

        color: colors[packageTemplates.length % colors.length]

      });

      renderPackageList();

    }


    function renderPackageList() {

      const list = document.getElementById('packageList');

      list.innerHTML = '';


      packageTemplates.forEach((pkg) => {

        const div = document.createElement('div');

        div.className = 'package-template';

        div.draggable = true;

        div.ondragstart = (event) => {

          event.dataTransfer.setData('text/plain', pkg.id);

        };

        div.innerHTML = `

          <div>

            <div class="package-title">${escapeHtml(pkg.name)} × ${pkg.count}</div>

            <div class="package-meta">${pkg.lengthCm} × ${pkg.widthCm} cm</div>

          </div>

          <button class="small-button danger" onclick="removeTemplate('${pkg.id}')">X</button>

        `;

        list.appendChild(div);

      });

    }


    function removeTemplate(id) {

      packageTemplates = packageTemplates.filter(p => p.id !== id);

      renderPackageList();

    }


    function dropOnTrailer(event) {

      event.preventDefault();

      const templateId = event.dataTransfer.getData('text/plain');

      const pkg = packageTemplates.find(p => p.id === templateId);

      if (!pkg) return;


      const trailer = document.getElementById('trailer');

      const rect = trailer.getBoundingClientRect();

      const x = snap(event.clientX - rect.left);

      const y = snap(event.clientY - rect.top);


      placeBox({

        templateId: pkg.id,

        name: pkg.name,

        lengthCm: pkg.lengthCm,

        widthCm: pkg.widthCm,

        x,

        y,

        color: pkg.color,

        rotated: false,

      });

    }


    function placeBox(data) {

      const box = {

        id: crypto.randomUUID(),

        ...data,

      };

      placedBoxes.push(box);

      selectedId = box.id;

      validateAllBoxes();

      renderPlacedBoxes();

      updateStats();

    }


    function renderPlacedBoxes() {

      const trailer = document.getElementById('trailer');

      [...trailer.querySelectorAll('.placed-box')].forEach(el => el.remove());


      placedBoxes.forEach(box => {

        const el = document.createElement('div');

        el.className = 'placed-box';

        if (box.id === selectedId) el.classList.add('selected');

        if (box.invalid) el.classList.add('invalid');

        el.style.left = `${box.x}px`;

        el.style.top = `${box.y}px`;

        el.style.width = `${boxWidthPx(box)}px`;

        el.style.height = `${boxHeightPx(box)}px`;

        el.style.background = box.invalid ? 'rgba(239,68,68,.78)' : box.color;

        el.innerHTML = `${escapeHtml(box.name)}<br>${box.rotated ? box.widthCm + '×' + box.lengthCm : box.lengthCm + '×' + box.widthCm} cm`;


        el.addEventListener('mousedown', (event) => startMove(event, box.id));

        el.addEventListener('click', (event) => {

          event.stopPropagation();

          selectedId = box.id;

          renderPlacedBoxes();

        });

        trailer.appendChild(el);

      });


      trailer.onclick = () => {

        selectedId = null;

        renderPlacedBoxes();

      };

    }


    function startMove(event, id) {

      event.preventDefault();

      event.stopPropagation();

      selectedId = id;

      const box = placedBoxes.find(b => b.id === id);

      dragState = {

        id,

        startMouseX: event.clientX,

        startMouseY: event.clientY,

        startX: box.x,

        startY: box.y,

      };

      document.addEventListener('mousemove', moveSelected);

      document.addEventListener('mouseup', stopMove);

      renderPlacedBoxes();

    }


    function moveSelected(event) {

      if (!dragState) return;

      const box = placedBoxes.find(b => b.id === dragState.id);

      box.x = snap(dragState.startX + (event.clientX - dragState.startMouseX));

      box.y = snap(dragState.startY + (event.clientY - dragState.startMouseY));

      validateAllBoxes();

      renderPlacedBoxes();

      updateStats();

    }


    function stopMove() {

      dragState = null;

      document.removeEventListener('mousemove', moveSelected);

      document.removeEventListener('mouseup', stopMove);

    }


    function rotateSelected() {

      const box = placedBoxes.find(b => b.id === selectedId);

      if (!box) return;

      box.rotated = !box.rotated;

      validateAllBoxes();

      renderPlacedBoxes();

      updateStats();

    }


    function deleteSelected() {

      if (!selectedId) return;

      placedBoxes = placedBoxes.filter(b => b.id !== selectedId);

      selectedId = null;

      renderPlacedBoxes();

      updateStats();

    }


    function clearPlaced() {

      placedBoxes = [];

      selectedId = null;

      renderPlacedBoxes();

      updateStats();

    }


    function autoPack() {

      const items = [];

      packageTemplates.forEach(pkg => {

        for (let i = 0; i < pkg.count; i++) {

          items.push({ ...pkg });

        }

      });


      items.sort((a, b) => (b.lengthCm * b.widthCm) - (a.lengthCm * a.widthCm));


      const tpx = trailerPx();

      const packed = [];

      const unpacked = [];


      for (const item of items) {

        const candidates = [

          { rotated: false, w: item.lengthCm, h: item.widthCm },

          { rotated: true, w: item.widthCm, h: item.lengthCm },

        ];


        let placed = false;


        for (let y = 0; y <= tpx.height && !placed; y += GRID_PX) {

          for (let x = 0; x <= tpx.width && !placed; x += GRID_PX) {

            for (const c of candidates) {

              const candidate = {

                id: crypto.randomUUID(),

                templateId: item.id,

                name: item.name,

                lengthCm: item.lengthCm,

                widthCm: item.widthCm,

                x,

                y,

                color: item.color,

                rotated: c.rotated,

              };

              if (fits(candidate, packed, tpx)) {

                packed.push(candidate);

                placed = true;

                break;

              }

            }

          }

        }


        if (!placed) unpacked.push(item);

      }


      placedBoxes = packed;

      selectedId = null;

      validateAllBoxes();

      renderPlacedBoxes();

      updateStats();


      if (unpacked.length) {

        alert(`${unpacked.length} Paket(e) konnten nicht platziert werden. Ladefläche ist rechnerisch voll oder die Maße passen nicht.`);

      }

    }


    function fits(candidate, boxes, tpx) {

      const cw = boxWidthPx(candidate);

      const ch = boxHeightPx(candidate);


      if (candidate.x < 0 || candidate.y < 0 || candidate.x + cw > tpx.width || candidate.y + ch > tpx.height) {

        return false;

      }


      return !boxes.some(other => overlaps(candidate, other));

    }


    function validateAllBoxes() {

      const tpx = trailerPx();

      placedBoxes.forEach(box => {

        box.invalid = !fits(box, placedBoxes.filter(b => b.id !== box.id), tpx);

      });

    }


    function overlaps(a, b) {

      return !(

        a.x + boxWidthPx(a) <= b.x ||

        b.x + boxWidthPx(b) <= a.x ||

        a.y + boxHeightPx(a) <= b.y ||

        b.y + boxHeightPx(b) <= a.y

      );

    }


    function boxWidthPx(box) {

      return box.rotated ? box.widthCm : box.lengthCm;

    }


    function boxHeightPx(box) {

      return box.rotated ? box.lengthCm : box.widthCm;

    }


    function snap(value) {

      return Math.max(0, Math.round(value / GRID_PX) * GRID_PX);

    }


    function updateStats() {

      const t = trailerMeters();

      const totalArea = t.length * t.width;

      const usedArea = placedBoxes.reduce((sum, b) => sum + (b.lengthCm * b.widthCm / 10000), 0);

      const usage = totalArea > 0 ? Math.round((usedArea / totalArea) * 100) : 0;
